Xác thực (Authentication) với JWT

Luồng chuẩn trong Nest 11: endpoint đăng nhập trả JWT, một guard global kiểm tra token ở mọi route, decorator @Public() mở các route không cần đăng nhập.

npm i @nestjs/jwt
@Module({
  imports: [
    UsersModule,
    JwtModule.registerAsync({
      global: true,
      inject: [ConfigService],
      useFactory: (c: ConfigService) => ({
        secret: c.getOrThrow('JWT_SECRET'),
        signOptions: { expiresIn: '15m' },
      }),
    }),
  ],
  controllers: [AuthController],
  providers: [AuthService, { provide: APP_GUARD, useClass: AuthGuard }],
})
export class AuthModule {}

@Injectable()
export class AuthService {
  constructor(private readonly users: UsersService, private readonly jwt: JwtService) {}

  async signIn(email: string, password: string) {
    const user = await this.users.findByEmail(email);
    if (!user || !(await argon2.verify(user.passwordHash, password))) {
      throw new UnauthorizedException('Sai email hoặc mật khẩu');
    }
    const payload = { sub: user.id, roles: user.roles };
    return { accessToken: await this.jwt.signAsync(payload) };
  }
}

@Injectable()
export class AuthGuard implements CanActivate {
  constructor(private readonly jwt: JwtService, private readonly reflector: Reflector) {}

  async canActivate(ctx: ExecutionContext) {
    const isPublic = this.reflector.getAllAndOverride(IS_PUBLIC, [ctx.getHandler(), ctx.getClass()]);
    if (isPublic) return true;

    const req = ctx.switchToHttp().getRequest();
    const [type, token] = req.headers.authorization?.split(' ') ?? [];
    if (type !== 'Bearer' || !token) throw new UnauthorizedException();

    try {
      req.user = await this.jwt.verifyAsync(token);   // gắn user vào request cho guard/handler sau
    } catch {
      throw new UnauthorizedException();
    }
    return true;
  }
}

@Controller('auth')
export class AuthController {
  constructor(private readonly auth: AuthService) {}

  @Public()
  @Post('login')
  @HttpCode(HttpStatus.OK)
  login(@Body() dto: LoginDto) {
    return this.auth.signIn(dto.email, dto.password);
  }

  @Get('profile')
  profile(@Req() req) {            // route mặc định đã được bảo vệ bởi APP_GUARD
    return req.user;
  }
}

Passport (@nestjs/passport) là lựa chọn khi cần nhiều chiến lược (local, JWT, OAuth Google/GitHub…): viết class kế thừa PassportStrategy(Strategy) với method validate(), rồi dùng AuthGuard('jwt').

Phân quyền (Authorization)

RBAC: decorator @Roles() + RolesGuard như ở mục Guards. Với quyền theo dữ liệu ("chỉ sửa bài của chính mình"), kiểm tra trong service vì guard chưa có bản ghi. Thư viện CASL cộng đồng giúp định nghĩa ability dạng can('update', 'Article', {'{'} authorId: user.id {'}'}).

@Injectable()
export class ArticlesService {
  async update(user: AuthUser, id: string, dto: UpdateArticleDto) {
    const article = await this.repo.findById(id);
    if (!article) throw new NotFoundException();
    if (article.authorId !== user.sub && !user.roles.includes('admin')) {
      throw new ForbiddenException();
    }
    return this.repo.update(id, dto);
  }
}
v12: @nestjs/authorization

Phân quyền theo policy class: @Policy() class ProductPolicy {'{'} update(user, product) {'{'} … {'}'} {'}'}, gắn bằng @Can(ProductPolicy, 'update'); mặc định từ chối mọi handler không khai báo.

WebSockets

npm i @nestjs/websockets @nestjs/platform-socket.io
@WebSocketGateway({ namespace: '/live', cors: { origin: '*' } })
export class LiveGateway implements OnGatewayConnection, OnGatewayDisconnect {
  @WebSocketServer() server: Namespace;

  handleConnection(client: Socket) {
    const shop = client.handshake.auth?.shop;
    client.join(`shop:${shop}`);
  }
  handleDisconnect(client: Socket) {}

  @SubscribeMessage('ping')
  ping(@MessageBody() data: { t: number }, @ConnectedSocket() client: Socket) {
    return { event: 'pong', data: { t: data.t } };   // trả WsResponse → gửi event 'pong'
  }

  pushVisitor(shop: string, v: Visitor) {           // gọi từ service khác
    this.server.to(`shop:${shop}`).emit('visitor', v);
  }
}
  • Lỗi ném bằng WsException; client nhận event exception. Filter global HTTP không áp dụng, dùng @UseFilters() trên gateway.
  • ValidationPipe ném BadRequestException, phải đặt exceptionFactory: (e) => new WsException(e).
  • handleConnection không chặn được kết nối; xác thực bằng guard hoặc middleware server.use() của socket.io.
  • Chạy nhiều instance: custom IoAdapter dùng @socket.io/redis-adapter + sticky session.

Microservices @nestjs/microservices

Cùng mô hình controller/DI nhưng thay HTTP bằng transport: TCP, Redis, RabbitMQ (RMQ), Kafka, NATS, gRPC, MQTT hoặc transport tự viết.

DecoratorKiểu giao tiếpPhía gửi
@MessagePattern()Request-response, có giá trị trả vềclient.send() trả Observable "lạnh", phải subscribe/await
@EventPattern()Fire-and-forget, nhiều handler chạy song songclient.emit()
const app = await NestFactory.createMicroservice(WorkerModule, {
  transport: Transport.RMQ,
  options: {
    urls: [process.env.AMQP_URL!],
    queue: 'orders',
    noAck: false,                         // ack thủ công
    prefetchCount: 20,
    queueOptions: { durable: true },
  },
});
await app.listen();

@Controller()
export class OrdersConsumer {
  @EventPattern('order.created')
  async onCreated(@Payload() data: OrderCreated, @Ctx() ctx: RmqContext) {
    const channel = ctx.getChannelRef();
    const msg = ctx.getMessage();
    try {
      await this.service.process(data);
      channel.ack(msg);
    } catch (e) {
      channel.nack(msg, false, false);     // không requeue → về dead-letter queue
    }
  }

  @MessagePattern({ cmd: 'risk.score' })
  score(@Payload() data: { orderId: string }) {
    return this.risk.score(data.orderId);   // giá trị trả về là response
  }
}

// phía gọi
@Module({
  imports: [ClientsModule.register([{ name: 'WORKER', transport: Transport.RMQ, options: { urls: [url], queue: 'orders' } }])],
})
export class ApiModule {}

@Injectable()
export class OrdersPublisher {
  constructor(@Inject('WORKER') private readonly client: ClientProxy) {}
  publish(o: OrderCreated) { return this.client.emit('order.created', o); }
  score(id: string) { return firstValueFrom(this.client.send({ cmd: 'risk.score' }, { orderId: id }).pipe(timeout(5000))); }
}

Hybrid app (vừa HTTP vừa nghe queue): app.connectMicroservice({'{'} transport, options {'}'}, {'{'} inheritAppConfig: true {'}'}), rồi await app.startAllMicroservices() và app.listen(). Lỗi trong handler RPC ném bằng RpcException.

GraphQL @nestjs/graphql

Hai cách tiếp cận: code-first (viết class TypeScript, Nest sinh schema) và schema-first (viết SDL, Nest sinh type). Code-first phổ biến hơn trong dự án TypeScript.

GraphQLModule.forRoot({
  driver: ApolloDriver,
  autoSchemaFile: true,       // sinh schema trong bộ nhớ (hoặc đường dẫn file)
  sortSchema: true,
});

@ObjectType()
export class Shop {
  @Field(() => ID) id: string;
  @Field() domain: string;
  @Field(() => Int, { nullable: true }) plan?: number;
}

@InputType()
export class UpdateShopInput {
  @Field() @IsString() domain: string;
}

@Resolver(() => Shop)
export class ShopResolver {
  constructor(private readonly shops: ShopsService) {}

  @Query(() => Shop, { name: 'shop' })
  getShop(@Args('id', { type: () => ID }) id: string) { return this.shops.find(id); }

  @Mutation(() => Shop)
  updateShop(@Args('input') input: UpdateShopInput) { return this.shops.update(input); }

  @ResolveField(() => [Rule])
  rules(@Parent() shop: Shop) { return this.rules.byShop(shop.id); }   // cẩn thận N+1 → dùng DataLoader
}

Guard/interceptor dùng GqlExecutionContext.create(ctx).getContext().req. Subscriptions dùng graphql-ws; PubSub có sẵn chỉ chạy trong bộ nhớ, production dùng Redis PubSub. Federation có ApolloFederationDriver / ApolloGatewayDriver.

CQRS @nestjs/cqrs

Tách thao tác ghi (Command) và đọc (Query) thành các handler riêng, giao tiếp qua bus. Hợp với domain phức tạp có luồng đọc/ghi khác hẳn nhau; với CRUD thông thường thì là thừa.

export class BlockIpCommand extends Command<{ ruleId: string }> {
  constructor(public readonly shop: string, public readonly ip: string) { super(); }
}

@CommandHandler(BlockIpCommand)
export class BlockIpHandler implements ICommandHandler {
  constructor(private readonly repo: RulesRepository, private readonly publisher: EventPublisher) {}

  async execute({ shop, ip }: BlockIpCommand) {
    const rule = this.publisher.mergeObjectContext(await this.repo.createBlock(shop, ip));
    rule.markCreated();          // AggregateRoot.apply(new IpBlockedEvent(...))
    rule.commit();               // phát event ra EventBus
    return { ruleId: rule.id };
  }
}

@EventsHandler(IpBlockedEvent)
export class SyncTrackerOnBlock implements IEventHandler {
  handle(e: IpBlockedEvent) { /* đẩy sang tracker */ }
}

// controller
const { ruleId } = await this.commandBus.execute(new BlockIpCommand(shop, ip));
const rules = await this.queryBus.execute(new ListRulesQuery(shop));
