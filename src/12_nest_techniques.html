Configuration @nestjs/config

Đọc biến môi trường từ .env và process.env, validate lúc khởi động, rồi truy cập qua ConfigService thay vì gọi process.env rải rác.

npm i @nestjs/config
import { registerAs } from '@nestjs/config';

// config theo namespace: gom biến liên quan vào một object có kiểu
export default registerAs('database', () => ({
  uri: process.env.MONGO_URI!,
  poolSize: parseInt(process.env.MONGO_POOL_SIZE ?? '10', 10),
}));

@Module({
  imports: [
    ConfigModule.forRoot({
      isGlobal: true,                       // không cần import lại ở module con
      envFilePath: ['.env.local', '.env'],  // file đầu tiên thắng
      load: [databaseConfig, redisConfig],
      cache: true,                          // cache giá trị đọc từ process.env
      validate: validateEnv,                // fail fast nếu thiếu biến
    }),
    MongooseModule.forRootAsync({
      inject: [databaseConfig.KEY],
      useFactory: (db: ConfigType) => ({ uri: db.uri, maxPoolSize: db.poolSize }),
    }),
  ],
})
export class AppModule {}

Validate biến môi trường bằng class-validator

import { plainToInstance } from 'class-transformer';
import { IsEnum, IsInt, IsString, validateSync } from 'class-validator';

class EnvVars {
  @IsEnum(['development', 'production', 'test']) NODE_ENV: string;
  @IsInt() PORT: number;
  @IsString() MONGO_URI: string;
}

export function validateEnv(config: Record) {
  const env = plainToInstance(EnvVars, config, { enableImplicitConversion: true });
  const errors = validateSync(env, { skipMissingProperties: false });
  if (errors.length) throw new Error(errors.toString());
  return env;
}

MongoDB với Mongoose @nestjs/mongoose

import { Prop, Schema, SchemaFactory } from '@nestjs/mongoose';
import { HydratedDocument, Types } from 'mongoose';

export type OrderDocument = HydratedDocument;

@Schema({ timestamps: true, collection: 'orders' })
export class Order {
  @Prop({ required: true, index: true }) shop: string;
  @Prop({ required: true }) orderId: string;
  @Prop({ type: [String], default: [] }) tags: string[];
  @Prop({ type: Types.ObjectId, ref: 'Customer' }) customer?: Types.ObjectId;
  @Prop({ type: Object }) riskSnapshot?: Record;
}

export const OrderSchema = SchemaFactory.createForClass(Order);
OrderSchema.index({ shop: 1, orderId: 1 }, { unique: true });   // compound index khai báo ở schema

// module
MongooseModule.forFeature([{ name: Order.name, schema: OrderSchema }]);

// service / repository
@Injectable()
export class OrdersRepository {
  constructor(
    @InjectModel(Order.name) private readonly model: Model,
    @InjectConnection() private readonly conn: Connection,
  ) {}

  upsert(shop: string, orderId: string, data: Partial) {
    return this.model.findOneAndUpdate({ shop, orderId }, { $set: data }, { upsert: true, new: true }).lean();
  }

  async transfer(a: string, b: string) {
    const session = await this.conn.startSession();
    try {
      await session.withTransaction(async () => {
        await this.model.updateOne({ _id: a }, { $inc: { credit: -1 } }, { session });
        await this.model.updateOne({ _id: b }, { $inc: { credit: 1 } }, { session });
      });
    } finally {
      await session.endSession();
    }
  }
}

Nhiều kết nối: đặt connectionName trong forRoot, rồi truyền tên đó vào forFeature(..., 'analytics') và @InjectModel(X.name, 'analytics'). Hook và plugin của schema dùng forFeatureAsync. Trong test, mock bằng getModelToken(Order.name).

ValidationPipe + class-validator

npm i class-validator class-transformer
import { Type } from 'class-transformer';
import {
  ArrayMaxSize, IsArray, IsBoolean, IsEnum, IsIP, IsOptional,
  IsString, Length, ValidateNested,
} from 'class-validator';

export enum RuleAction { BLOCK = 'block', ALLOW = 'allow', REDIRECT = 'redirect' }

class Condition {
  @IsString() field: string;
  @IsString() value: string;
}

export class CreateRuleDto {
  @IsString() @Length(1, 120) name: string;
  @IsEnum(RuleAction) action: RuleAction;
  @IsOptional() @IsIP() ip?: string;
  @IsBoolean() enabled = true;                       // giá trị mặc định (cần transform: true)

  @IsArray() @ArrayMaxSize(50)
  @ValidateNested({ each: true }) @Type(() => Condition)   // validate object lồng nhau
  conditions: Condition[];
}

app.useGlobalPipes(new ValidationPipe({
  whitelist: true,              // bỏ các field không có decorator
  forbidNonWhitelisted: true,   // ...hoặc trả 400 nếu client gửi field lạ
  transform: true,              // chuyển payload thành instance DTO, ép kiểu param/query
  transformOptions: { enableImplicitConversion: true },
  stopAtFirstError: true,
  // exceptionFactory: (errors) => new UnprocessableEntityException(errors),
}));

Serialization: kiểm soát dữ liệu trả về

ClassSerializerInterceptor chạy instanceToPlain của class-transformer trên giá trị trả về. Handler phải trả về instance của class (hoặc khai báo @SerializeOptions({'{'} type {'}'})).

export class UserEntity {
  id: string;
  email: string;
  @Exclude() password: string;                                 // không bao giờ trả ra
  @Expose() get displayName() { return this.email.split('@')[0]; }
  @Transform(({ value }) => value?.name) role: Role;           // chỉ trả role.name
  constructor(partial: Partial) { Object.assign(this, partial); }
}

@UseInterceptors(ClassSerializerInterceptor)
@SerializeOptions({ excludeExtraneousValues: false })
@Get(':id')
async find(@Param('id') id: string) {
  return new UserEntity(await this.users.findOne(id));
}

Caching @nestjs/cache-manager

Từ Nest 11, cache-manager v6 dùng Keyv làm tầng lưu trữ. TTL tính bằng mili-giây.

npm i @nestjs/cache-manager cache-manager keyv @keyv/redis cacheable
import { CacheModule } from '@nestjs/cache-manager';
import { createKeyv } from '@keyv/redis';
import { Keyv } from 'keyv';
import { CacheableMemory } from 'cacheable';

CacheModule.registerAsync({
  isGlobal: true,
  useFactory: () => ({
    ttl: 60_000,
    stores: [
      new Keyv({ store: new CacheableMemory({ ttl: 60_000, lruSize: 5000 }) }), // L1 in-memory
      createKeyv('redis://localhost:6379'),                                    // L2 Redis
    ],
  }),
});

@Injectable()
export class SettingsService {
  constructor(@Inject(CACHE_MANAGER) private readonly cache: Cache) {}

  async get(shop: string) {
    const key = `settings:${shop}`;
    const cached = await this.cache.get(key);   // miss → undefined
    if (cached) return cached;
    const fresh = await this.repo.findByShop(shop);
    await this.cache.set(key, fresh, 5 * 60_000);
    return fresh;
  }

  async update(shop: string, dto: UpdateSettingsDto) {
    await this.repo.update(shop, dto);
    await this.cache.del(`settings:${shop}`);             // cache-aside: xóa khi ghi
  }
}

Cache tự động cho route GET: @UseInterceptors(CacheInterceptor), tinh chỉnh bằng @CacheKey('k') và @CacheTTL(30_000). Muốn key theo shop/user, kế thừa CacheInterceptor và override trackBy(context).

Cạm bẫy
  • Store Keyv serialize sang JSON: Date lấy ra thành chuỗi.
  • Store in-memory giữ tham chiếu object: sửa object lấy từ cache là sửa luôn cache.
  • Luôn gắn listener 'error' cho store Redis, nếu không mất kết nối có thể làm sập app.

Task scheduling @nestjs/schedule

@Module({ imports: [ScheduleModule.forRoot()] })    // chỉ import ở MỘT nơi, nếu không job chạy nhiều lần
export class AppModule {}

@Injectable()
export class CleanupJobs {
  private readonly logger = new Logger(CleanupJobs.name);
  constructor(private readonly registry: SchedulerRegistry) {}

  @Cron(CronExpression.EVERY_DAY_AT_2AM, { name: 'purge-visitors', timeZone: 'Asia/Ho_Chi_Minh', waitForCompletion: true })
  async purge() { /* ... */ }

  @Cron('0 */15 * * * *')            // giây phút giờ ngày tháng thứ → mỗi 15 phút
  syncQuota() {}

  @Interval('heartbeat', 30_000) heartbeat() {}
  @Timeout(5_000) warmup() {}        // chạy một lần sau 5 giây

  pause() {
    this.registry.getCronJob('purge-visitors').stop();
  }

  addDynamic(name: string, expr: string) {
    const job = new CronJob(expr, () => this.logger.log(`run ${name}`));
    this.registry.addCronJob(name, job);
    job.start();
  }
}

Queues với BullMQ @nestjs/bullmq

Đưa việc nặng (gửi mail, gọi API chậm, xử lý file) ra khỏi request: producer đẩy job vào Redis, worker xử lý với retry/backoff.

npm i @nestjs/bullmq bullmq ioredis
@Module({
  imports: [
    BullModule.forRoot({ connection: { host: 'localhost', port: 6379 }, defaultJobOptions: { removeOnComplete: 1000 } }),
    BullModule.registerQueue({ name: 'webhooks' }),
  ],
  providers: [WebhookProducer, WebhookConsumer],
})
export class WebhooksModule {}

// producer
@Injectable()
export class WebhookProducer {
  constructor(@InjectQueue('webhooks') private readonly queue: Queue) {}

  enqueue(payload: OrderCreated) {
    return this.queue.add('order-created', payload, {
      jobId: `order:${payload.id}`,                   // chống trùng job
      attempts: 5,
      backoff: { type: 'exponential', delay: 2_000 },
    });
  }
}

// consumer
@Processor('webhooks', { concurrency: 10 })
export class WebhookConsumer extends WorkerHost {
  async process(job: Job) {
    switch (job.name) {                               // BullMQ không có @Process('name')
      case 'order-created': return this.handleOrder(job.data);
    }
  }

  @OnWorkerEvent('failed')
  onFailed(job: Job, err: Error) { /* log, alert */ }
}

Job định kỳ trong BullMQ dùng queue.upsertJobScheduler('id', {'{'} pattern: '0 * * * *' {'}'}) (option repeat cũ đã bỏ). Lắng nghe event toàn queue bằng @QueueEventsListener('webhooks').

Events nội bộ @nestjs/event-emitter

Event emitter giúp tách module: OrdersService chỉ phát event "order.created", các module khác tự lắng nghe mà OrdersService không cần biết.

EventEmitterModule.forRoot({ wildcard: true, delimiter: '.' });

export class OrderCreatedEvent { constructor(public readonly shop: string, public readonly orderId: string) {} }

@Injectable()
export class OrdersService {
  constructor(private readonly events: EventEmitter2) {}
  async create(dto: CreateOrderDto) {
    const order = await this.repo.create(dto);
    this.events.emit('order.created', new OrderCreatedEvent(order.shop, order.id));
    return order;
  }
}

@Injectable()
export class RiskListener {
  @OnEvent('order.created', { async: true })
  async score(e: OrderCreatedEvent) { /* ... */ }

  @OnEvent('order.*')                 // wildcard 1 cấp; '**' nhiều cấp
  audit(e: unknown) {}
}
Giới hạn

Event nằm trong bộ nhớ của một process: process chết là mất, không retry, không chia sang instance khác. Mặc định lỗi trong listener bị nuốt (suppressErrors: true). Việc quan trọng thì dùng queue hoặc message broker kèm Outbox.

Các mục nhỏ khác
